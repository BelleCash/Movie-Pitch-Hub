## CREATE UTILITY

```js
const calculateTrendingScore = (pitch) => {
  const likesWeight = pitch.likes * 3;

  const freshnessWeight =
    Math.max(
      0,
      100 -
      ((Date.now() - new Date(pitch.created_at)) / 3600000)
    );

  const tagBonus =
    (pitch.tags?.length || 0) * 5;

  return likesWeight + freshnessWeight + tagBonus;
};
```
